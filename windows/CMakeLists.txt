cmake_minimum_required(VERSION 3.21)
project(OpenstratorWindows VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(openstrator_core INTERFACE)
target_include_directories(openstrator_core INTERFACE src)
if (MSVC)
  target_compile_options(openstrator_core INTERFACE /W4 /permissive- /EHsc /utf-8)
else()
  target_compile_options(openstrator_core INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(openstrator_core_tests tests/core_tests.cpp)
target_link_libraries(openstrator_core_tests PRIVATE openstrator_core)
enable_testing()
add_test(NAME openstrator_core COMMAND openstrator_core_tests)
if (WIN32)
  add_executable(Openstrator WIN32 src/main.cpp)
  target_link_libraries(Openstrator PRIVATE openstrator_core d2d1 dwrite comdlg32 ole32 user32 gdi32)
  set_target_properties(Openstrator PROPERTIES OUTPUT_NAME "Openstrator")
  if (MSVC)
    target_link_options(Openstrator PRIVATE /SUBSYSTEM:WINDOWS)
  endif()
endif()
